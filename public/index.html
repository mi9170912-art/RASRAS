/** * Simple Order & Product Management API * File: index.js */ const express = require('express'); const app = express(); const PORT = process.env.PORT || 3000; // Middleware to parse JSON bodies app.use(express.json()); // In-Memory Database let products = [ { id: 1, name: 'Product A', price: 100, stock: 50 }, { id: 2, name: 'Product B', price: 250, stock: 20 } ]; let orders = []; // Base Route app.get('/', (req, res) => { res.json({ message: 'Welcome to the Order Management API', status: 'Running', endpoints: { getProducts: 'GET /api/products', createProduct: 'POST /api/products', getOrders: 'GET /api/orders', createOrder: 'POST /api/orders' } }); }); // 1. Get All Products app.get('/api/products', (req, res) => { res.status(200).json({ success: true, data: products }); }); // 2. Add New Product app.post('/api/products', (req, res) => { const { name, price, stock } = req.body; if (!name || price == null || stock == null) { return res.status(400).json({ success: false, message: 'Please provide name, price, and stock.' }); } const newProduct = { id: products.length + 1, name, price: Number(price), stock: Number(stock) }; products.push(newProduct); res.status(201).json({ success: true, data: newProduct }); }); // 3. Create New Order app.post('/api/orders', (req, res) => { const { productId, quantity } = req.body; const product = products.find(p => p.id === Number(productId)); if (!product) { return res.status(404).json({ success: false, message: 'Product not found' }); } if (product.stock < quantity) { return res.status(400).json({ success: false, message: 'Insufficient stock' }); } // Deduct stock product.stock -= Number(quantity); const newOrder = { id: orders.length + 1, productId: product.id, productName: product.name, quantity: Number(quantity), totalPrice: product.price * Number(quantity), createdAt: new Date().toISOString() }; orders.push(newOrder); res.status(201).json({ success: true, data: newOrder }); }); // 4. Get All Orders app.get('/api/orders', (req, res) => { res.status(200).json({ success: true, data: orders }); }); // Global Error Handler app.use((err, req, res, next) => { console.error(err.stack); res.status(500).json({ success: false, message: 'Internal Server Error' }); }); // Start Server app.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
